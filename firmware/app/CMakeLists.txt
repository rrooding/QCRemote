cmake_minimum_required(VERSION 3.20.0)

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(qc_bridge_firmware)

# ETL (https://github.com/ETLCPP/etl) - fixed-capacity containers, no heap
# allocation. See CODING_STANDARDS.md's Firmware/Memory & containers section.
include(FetchContent)
FetchContent_Declare(
  etl
  GIT_REPOSITORY https://github.com/ETLCPP/etl.git
  GIT_TAG        20.49.0
)
FetchContent_MakeAvailable(etl)

# miniz (https://github.com/richgel999/miniz) - issue #6's gzip detection
# for compressed message payloads, see ADR 0004. We want only its
# low-level inflate function (tinfl_decompress), not the deflate/zip-
# archive/zlib-wrapper code the rest of miniz also provides, and not its
# own CMakeLists.txt (which builds all of that as one target and calls
# project() in a way that doesn't nest cleanly inside a Zephyr app) - so
# this fetches the source tree only (FetchContent_Populate, not
# MakeAvailable: deliberately does not add_subdirectory it) and compiles
# just miniz_tinfl.c ourselves. The MINIZ_NO_* defines below strip the
# declarations miniz.h would otherwise pull in for the parts we're not
# compiling down to nothing. miniz_export.h (normally generated by
# miniz's own CMake via generate_export_header() for a shared-library
# build) is our own trivial stub - see third_party/miniz/miniz_export.h.
FetchContent_Declare(
  miniz
  GIT_REPOSITORY https://github.com/richgel999/miniz.git
  GIT_TAG        3.0.2
)
FetchContent_Populate(miniz)

# subsys/usb/host's own headers (usbh_class.h, usbh_desc.h, ...) aren't part
# of Zephyr's public API and aren't exported by its build - needed by
# usbh_shim.c, which is the only file that includes them. See that file.
zephyr_include_directories(${ZEPHYR_BASE}/subsys/usb/host)

target_sources(app PRIVATE
  src/Main.cpp
  src/usbh_shim.c
  ${miniz_SOURCE_DIR}/miniz_tinfl.c
)
target_include_directories(app PRIVATE
  ${miniz_SOURCE_DIR}
  ${CMAKE_CURRENT_SOURCE_DIR}/third_party/miniz
)
target_compile_definitions(app PRIVATE
  MINIZ_NO_STDIO
  MINIZ_NO_MALLOC
  MINIZ_NO_TIME
  # Not also defining MINIZ_NO_ARCHIVE_WRITING_APIS: miniz.h already
  # implies it from MINIZ_NO_DEFLATE_APIS (and MINIZ_NO_ARCHIVE_APIS
  # implies it a second time) - passing it here too just redefines the
  # same macro and trips a -Wmacro-redefined warning for no benefit.
  MINIZ_NO_DEFLATE_APIS
  MINIZ_NO_ARCHIVE_APIS
  MINIZ_NO_ZLIB_APIS
)
target_link_libraries(app PRIVATE etl::etl)
